Эффект мокрого асфальта build_height = random.uniform(10, 20) self.build_left = Entity( parent=self, model='cube', color=brick, scale=(5, build_height, CHUNK_LENGTH), position=(-8.5, build_height/2, 0), collider='box' ) self.build_right = Entity( parent=self, model='cube', color=brick, scale=(5, build_height, CHUNK_LENGTH), position=(8.5, build_height/2, 0), collider='box' ) self.pole = Entity( parent=self, model='cube', color=metal, scale=(0.2, 4.5, 0.2), position=(5.2, 2.25, 0) ) self.lamp = Entity( parent=self, model='sphere', color=amber_glow, scale=0.5, position=(5.2, 4.6, 0), shader=lit_with_shadows_shader ) self.lamp.material.set_emission(2.5) self.light = PointLight( parent=self, color=amber_glow, position=(5.2, 4.3, 0), attenuation=(1, 0.1, 0.01) ) lanes = [-1, 0, 1] random.shuffle(lanes) obs_lane = lanes[0] clue_lane = lanes[1] if random.random() < 0.5: obs = Entity( parent=self, model='cube', color=wood, scale=(2, 0.8, 1.5), position=(obs_lane * LANE_WIDTH, 0.4, 0), collider='box', obstacle_type='low' ) else: # Высокое (стена/авто) obs = Entity( parent=self, model='cube', color=metal, scale=(2.2, 2.2, 4), position=(obs_lane * LANE_WIDTH, 1.1, 0), collider='box', obstacle_type='high' ) self.obstacles.append(obs) # Улика (золотая монета) clue = Entity( parent=self, model='cube', color=gold_clue, scale=(0.6, 0.6, 0.1), position=(clue_lane * LANE_WIDTH, 1.2, 0), collider='sphere', active=True, rot_speed=random.uniform(2, 4) ) clue.material.set_emission(1.5) self.clues.append(clue) def spawn_chunk(): global last_chunk_z z_pos = last_chunk_z - CHUNK_LENGTH last_chunk_z = z_pos chunk = Chunk(z_pos) chunk.position = (0, 0, z_pos) chunks.append(chunk) for i in range(5): spawn_chunk() particles = [] def spawn_crash_particles(position): colors = [color.rgb(150, 160, 180), color.rgb(220, 220, 220), color.rgb(80, 50, 30)] for _ in range(40): p = Entity( model='cube', color=random.choice(colors), scale=random.uniform(0.05, 0.15), position=position + Vec3(0, 1, 0), velocity=Vec3(random.uniform(-3, 3), random.uniform(2, 5), random.uniform(-3, 3)), life=1.5 ) particles.append(p) def update_particles(): for p in particles[:]: p.life -= time.dt if p.life <= 0: particles.remove(p) destroy(p) continue p.velocity.y += GRAVITY * 0.5 * time.dt p.position += p.velocity * time.dt p.scale *= 0.9 def input(key): global current_lane, target_x, is_playing if not is_playing: return if key == 'left arrow' or key == 'a': current_lane = max(-1, current_lane - 1) elif key == 'right arrow' or key == 'd': current_lane = min(1, current_lane + 1) elif (key == 'up arrow' or key == 'w' or key == 'space') and not player.is_jumping: player.velocity_y = JUMP_FORCE player.is_jumping = True target_x = current_lane * LANE_WIDTH # --- UI --- score_text = Text( text='УЛИКИ: 0', position=(-0.85, 0.45), scale=2, color=color.gold, font='VCR_OSD_MONO.ttf' ) start_screen = Entity(parent=camera) Text( parent=start_screen, text='НУАРНЫЙ БЕГ', position=(0, 0.1), scale=3, color=color.orange, font='VCR_OSD_MONO.ttf' ) Text( parent=start_screen, text='Город хранит тайны. Собери улики, избегай мафии.\n← → смена полосы | ↑ или Пробел (прыжок)', position=(0, -0.05), scale=1.5, color=color.gray ) start_btn = Button( parent=start_screen, text='Начать расследование', color=color.dark_gray, position=(0, -0.25), scale=(0.4, 0.1) ) game_over_screen = Entity(parent=camera, enabled=False) Text( parent=game_over_screen, text='ДЕЛО ЗАКРЫТО', position=(0, 0.1), scale=3, color=color.red, font='VCR_OSD_MONO.ttf' ) final_score_text = Text( parent=game_over_screen, text='Собрано улик: 0', position=(0, -0.05), scale=2, color=color.white ) restart_btn = Button( parent=game_over_screen, text='Новое расследование', color=color.dark_gray, position=(0, -0.25), scale=(0.4, 0.1) ) def start_game(): global is_playing, score, game_speed, current_lane, target_x is_playing = True score = 0 game_speed = BASE_SPEED current_lane = 0 target_x = 0 player.position = (0, 0.8, 0) score_text.text = 'УЛИКИ: 0' start_screen.enabled = False game_over_screen.enabled = False for chunk in chunks[:]: destroy(chunk) chunks.clear() global last_chunk_z last_chunk_z = 0 for i in range(5): spawn_chunk() def game_over(): global is_playing is_playing = False spawn_crash_particles(player.position) game_over_screen.enabled = True final_score_text.text = f'Собрано улик: {score}' start_btn.on_click = start_game restart_btn.on_click = start_game def update(): global game_speed, score, last_chunk_z if not is_playing: return player.z -= game_speed * time.dt game_speed = min(MAX_SPEED, game_speed + 0.1 * time.dt) if last_chunk_z > player.z - CHUNK_LENGTH * 3: spawn_chunk() for chunk in chunks[:]: if chunk.z_pos > player.z + 20: chunks.remove(chunk) destroy(chunk) for chunk in chunks: for obs in chunk.obstacles: if player.intersects(obs).hit: if obs.obstacle_type == 'low' and player.y > 1.5: continue # Успешный перепрыг game_over() return for clue in chunk.clues: if not clue.active: continue clue.rotation_y += clue.rot_speed * time.dt if player.intersects(clue).hit: clue.active = False destroy(clue) score += 10 score_text.text = f'УЛИКИ: {score}' update_particles() def late_update(): camera.position = (player.x * 0.3, 4.5, player.z + 9) camera.look_at(player.position + Vec3(0, 1, -10)) app.run()
